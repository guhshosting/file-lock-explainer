cmake_minimum_required(VERSION 3.25)
project(file-lock-explainer VERSION 1.0.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "Windows and an MSVC-compatible compiler are required")
endif()
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_BINARY_DIR)
  message(FATAL_ERROR "Use an out-of-source build")
endif()
option(FLE_ANALYZE "Enable MSVC static analysis" OFF)
option(FLE_ASAN "Enable AddressSanitizer" OFF)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
configure_file(src/version.h.in generated/version.h @ONLY)
add_library(fle_core src/core.cpp src/format.cpp)
target_include_directories(fle_core PUBLIC src "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_compile_definitions(fle_core PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00 WINVER=0x0A00)
target_link_libraries(fle_core PUBLIC Rstrtmgr)
add_executable(file-lock-explainer src/main.cpp src/app.manifest)
target_link_libraries(file-lock-explainer PRIVATE fle_core)
include(CTest)
if(BUILD_TESTING)
  add_executable(unit-tests tests/unit.cpp)
  target_link_libraries(unit-tests PRIVATE fle_core)
  add_test(NAME unit COMMAND unit-tests)
  set_tests_properties(unit PROPERTIES TIMEOUT 30)
  add_executable(lock-helper tests/lock_helper.cpp)
  add_executable(rm-probe tests/rm_probe.cpp)
  target_link_libraries(rm-probe PRIVATE fle_core)
  find_program(NODE_EXECUTABLE node REQUIRED)
  add_test(NAME integration COMMAND "${NODE_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.js" "$<TARGET_FILE:file-lock-explainer>" "$<TARGET_FILE:lock-helper>" "${CMAKE_CURRENT_SOURCE_DIR}")
  set_tests_properties(integration PROPERTIES TIMEOUT 90)
  if(FLE_ASAN)
    get_filename_component(compiler_dir "${CMAKE_CXX_COMPILER}" DIRECTORY)
    set_tests_properties(unit integration PROPERTIES ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:${compiler_dir}")
  endif()
endif()
get_directory_property(project_targets BUILDSYSTEM_TARGETS)
foreach(t IN LISTS project_targets)
  get_target_property(target_type ${t} TYPE)
  if(target_type STREQUAL "UTILITY")
    continue()
  endif()
  target_compile_options(${t} PRIVATE /W4 /WX /utf-8 /permissive- /EHsc)
  if(FLE_ANALYZE)
    target_compile_options(${t} PRIVATE /analyze)
  endif()
  if(FLE_ASAN)
    target_compile_options(${t} PRIVATE /fsanitize=address /Zi)
    target_link_options(${t} PRIVATE /INCREMENTAL:NO /DEBUG)
  endif()
endforeach()
install(TARGETS file-lock-explainer RUNTIME DESTINATION bin)
install(FILES README.md LICENSE DESTINATION share/file-lock-explainer)
install(DIRECTORY docs DESTINATION share/file-lock-explainer)
